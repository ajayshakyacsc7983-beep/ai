import os
import threading
from flask import Flask, request, jsonify
from flask_cors import CORS
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

# --- AI का अपना दिमाग लोड करना (Microsoft DialoGPT) ---
print("AI का दिमाग लोड हो रहा है... इसमें थोड़ा समय लग सकता है...")
tokenizer = AutoTokenizer.from_pretrained("microsoft/DialoGPT-small")
model = AutoModelForCausalLM.from_pretrained("microsoft/DialoGPT-small")

# 2. Flask App सेटअप
app = Flask(__name__)
CORS(app)

@app.route('/api/chat', methods=['POST'])
def chat():
    data = request.json
    user_msg = data.get('message', '')

    try:
        # AI को सोचने और जवाब तैयार करने का प्रोसेस
        new_user_input_ids = tokenizer.encode(user_msg + tokenizer.eos_token, return_tensors='pt')
        
        # जवाब जनरेट करना
        chat_history_ids = model.generate(
            new_user_input_ids, 
            max_length=1000, 
            pad_token_id=tokenizer.eos_token_id,
            no_repeat_ngram_size=3,
            do_sample=True, 
            top_k=100, 
            top_p=0.7,
            temperature=0.8
        )
        
        ai_reply = tokenizer.decode(chat_history_ids[:, new_user_input_ids.shape[-1]:][0], skip_special_tokens=True)
        
        if not ai_reply:
            ai_reply = "Hmm, I'm not sure what to say to that."
            
    except Exception as e:
        ai_reply = "Error: " + str(e)

    return jsonify({"reply": ai_reply})

# 3. सर्वर को बैकग्राउंड में चलाना
def run_flask():
    os.system("fuser -k 5000/tcp")
    app.run(port=5000)

threading.Thread(target=run_flask).start()
